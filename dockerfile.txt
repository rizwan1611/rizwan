# DOCKER

Install:
sudo apt install docker.io -y

Start:
sudo systemctl start docker

Enable:
sudo systemctl enable docker

Version:
docker --version

Images:
docker images

Running Containers:
docker ps

All Containers:
docker ps -a

Pull Image:
docker pull ubuntu

Run Container:
docker run ubuntu

Interactive Container:
docker run -it ubuntu bash

Stop Container:
docker stop CONTAINER_ID

Delete Container:
docker rm CONTAINER_ID

Delete Image:
docker rmi IMAGE_ID

Build Image:
docker build -t myapp .

Run Image:
docker run myapp

# DOCKERFILE

FROM ubuntu

RUN apt update

CMD ["echo","Hello World"]

FROM = Base Image
RUN = Execute During Build
CMD = Execute When Container Starts
